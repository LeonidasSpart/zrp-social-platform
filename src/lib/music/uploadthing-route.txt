  musicTrack: f({
    audio: {

    .middleware(async () => {
      const session = await getServerSession(authOptions);
      if (!session?.user) throw new UploadThingError("Unauthorized");

    })
    .onUploadComplete(async ({ file }) => {
      return {

        type: file.type.startsWith("audio/") ? "audio" : "image",
